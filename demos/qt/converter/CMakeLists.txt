cmake_minimum_required(VERSION 3.22)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" APP_VERSION LIMIT_COUNT 1)
project(PapngConverter VERSION ${APP_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INSTALL_BINDIR ".")
set(CMAKE_INSTALL_LIBDIR "lib")
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Concurrent)
qt_standard_project_setup(REQUIRES 6.10)
qt_add_executable(papng-converter src/main.cpp src/cli.cpp src/cli.h src/controller.cpp src/controller.h src/pipeline.cpp src/pipeline.h src/writer.cpp)
target_link_libraries(papng-converter PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent)
target_compile_definitions(papng-converter PRIVATE PAPNG_CONVERTER_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(papng-converter PRIVATE /utf-8 /W4)
else()
    target_compile_options(papng-converter PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_add_qml_module(papng-converter URI PapngConverter VERSION 1.0 QML_FILES qml/Main.qml qml/Preview.qml qml/NumberField.qml)
set_target_properties(papng-converter PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
install(TARGETS papng-converter RUNTIME DESTINATION .)
# Select plugin file names, including Qt builds without imported plugin targets.
set(deploy_plugins qoffscreen)
if(UNIX)
    list(APPEND deploy_plugins qwayland qwayland-egl)
endif()
qt_generate_deploy_qml_app_script(TARGET papng-converter OUTPUT_SCRIPT deploy_script
    INCLUDE_PLUGINS ${deploy_plugins})
install(SCRIPT ${deploy_script})
